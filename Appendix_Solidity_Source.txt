pragma solidity ^0.8.24;

import "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

/**
 * @title MockPaymentToken
 * @author Propx Protocol
 * @notice Mock ERC-20 token used as payment currency in the Propx marketplace.
 *         Represents a simulated INR-equivalent stablecoin for prototype/demo purposes.
 *
 * @dev PROTOTYPE NOTICE:
 *      This is NOT a real stablecoin or regulated payment instrument.
 *      In production, an actual regulated stablecoin, CBDC, or payment rail would replace this.
 *      1 mINR token = 1 unit of simulated Indian Rupee (18 decimals).
 */
contract MockPaymentToken is ERC20, Ownable {
    /// @notice Maximum tokens mintable via public faucet per call
    uint256 public constant MAX_FAUCET_AMOUNT = 10_000_000 * 10 ** 18;

    event FaucetMinted(address indexed recipient, uint256 amount);

    constructor() ERC20("Mock INR Stablecoin", "mINR") Ownable(msg.sender) {
        // Mint 100 million mINR to deployer for initial distribution
        _mint(msg.sender, 100_000_000 * 10 ** 18);
    }

    /**
     * @notice Public faucet: anyone can mint tokens for testing.
     * @param amount Amount to mint in wei (18 decimals)
     */
    function faucet(uint256 amount) external {
        require(amount > 0, "MockPaymentToken: amount must be > 0");
        require(amount <= MAX_FAUCET_AMOUNT, "MockPaymentToken: exceeds faucet limit");
        _mint(msg.sender, amount);
        emit FaucetMinted(msg.sender, amount);
    }

    /**
     * @notice Owner can mint arbitrary amounts (for test setup scripts).
     */
    function mint(address to, uint256 amount) external onlyOwner {
        require(to != address(0), "MockPaymentToken: mint to zero address");
        _mint(to, amount);
    }
}

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@openzeppelin/contracts/access/AccessControl.sol";

/**
 * @title InvestorIdentity
 * @author Propx Protocol
 * @notice Manages permissioned investor eligibility for the Propx platform.
 *
 * @dev PRIVACY NOTICE:
 *      This contract does NOT store any personally identifiable information (PII)
 *      such as Aadhaar numbers, PAN details, passport numbers, or identity documents.
 *      Only wallet addresses and eligibility status are stored on-chain.
 *      Actual KYC verification is performed off-chain by a regulated KYC provider.
 *      An optional cryptographic reference hash may link the off-chain attestation
 *      to this on-chain record — this hash does not reveal any personal data.
 *
 * @dev PROTOTYPE NOTICE:
 *      KYC in this prototype is simulated. An admin approves investors manually.
 *      In production, integrate a compliant KYC oracle or attestation service.
 */
contract InvestorIdentity is AccessControl {
    bytes32 public constant KYC_ADMIN_ROLE   = keccak256("KYC_ADMIN_ROLE");
    bytes32 public constant KYC_ORACLE_ROLE  = keccak256("KYC_ORACLE_ROLE");

    /// @notice Investor verification status
    enum InvestorStatus {
        None,       // 0: Not registered
        Pending,    // 1: Registered, awaiting verification
        Approved,   // 2: KYC approved, eligible to invest
        Rejected,   // 3: KYC rejected
        Suspended   // 4: Previously approved, now suspended
    }

    /// @notice On-chain investor record (no PII stored)
    struct InvestorRecord {
        InvestorStatus status;
        uint256 registeredAt;
        uint256 approvedAt;
        uint256 updatedAt;
        bytes32 kycReferenceHash; // Off-chain KYC reference (not PII, just a pointer)
        string  rejectionReason;  // Human-readable, no PII
    }

    mapping(address => InvestorRecord) private _investors;
    address[] private _investorList;

    uint256 public totalRegistered;
    uint256 public totalApproved;
    uint256 public totalRejected;
    uint256 public totalSuspended;

    // ─── Events ──────────────────────────────────────────────────────────────
    event InvestorRegistered(address indexed investor, uint256 timestamp);
    event InvestorApproved(address indexed investor, address indexed approvedBy, uint256 timestamp);
    event InvestorRejected(address indexed investor, address indexed rejectedBy, string reason, uint256 timestamp);
    event InvestorSuspended(address indexed investor, address indexed suspendedBy, string reason, uint256 timestamp);
    event InvestorReinstated(address indexed investor, address indexed reinstatedBy, uint256 timestamp);
    event KycReferenceUpdated(address indexed investor, bytes32 referenceHash);

    // ─── Constructor ─────────────────────────────────────────────────────────
    constructor(address admin) {
        require(admin != address(0), "InvestorIdentity: admin is zero address");
        _grantRole(DEFAULT_ADMIN_ROLE, admin);
        _grantRole(KYC_ADMIN_ROLE,     admin);
        _grantRole(KYC_ORACLE_ROLE,    admin);
    }

    // ─── Investor Self-Registration ──────────────────────────────────────────

    /**
     * @notice Register your wallet address for KYC review. No PII is submitted on-chain.
     */
    function registerInvestor() external {
        require(
            _investors[msg.sender].status == InvestorStatus.None,
            "InvestorIdentity: already registered"
        );
        _investors[msg.sender] = InvestorRecord({
            status:           InvestorStatus.Pending,
            registeredAt:     block.timestamp,
            approvedAt:       0,
            updatedAt:        block.timestamp,
            kycReferenceHash: bytes32(0),
            rejectionReason:  ""
        });
        _investorList.push(msg.sender);
        totalRegistered++;
        emit InvestorRegistered(msg.sender, block.timestamp);
    }

    // ─── Admin Operations ────────────────────────────────────────────────────

    /**
     * @notice Approve an investor after off-chain KYC passes.
     * @param investor         Wallet to approve
     * @param kycReferenceHash Optional hash of off-chain KYC record (no PII)
     */
    function approveInvestor(address investor, bytes32 kycReferenceHash)
        external
        onlyRole(KYC_ADMIN_ROLE)
    {
        require(investor != address(0), "InvestorIdentity: zero address");
        InvestorRecord storage r = _investors[investor];
        require(
            r.status == InvestorStatus.Pending || r.status == InvestorStatus.Rejected,
            "InvestorIdentity: not in pending/rejected state"
        );
        r.status           = InvestorStatus.Approved;
        r.approvedAt       = block.timestamp;
        r.updatedAt        = block.timestamp;
        r.kycReferenceHash = kycReferenceHash;
        r.rejectionReason  = "";
        totalApproved++;
        emit InvestorApproved(investor, msg.sender, block.timestamp);
        if (kycReferenceHash != bytes32(0)) {
            emit KycReferenceUpdated(investor, kycReferenceHash);
        }
    }

    /**
     * @notice Reject an investor's KYC application.
     * @param investor Wallet to reject
     * @param reason   Rejection reason (no PII)
     */
    function rejectInvestor(address investor, string calldata reason)
        external
        onlyRole(KYC_ADMIN_ROLE)
    {
        require(investor != address(0), "InvestorIdentity: zero address");
        InvestorRecord storage r = _investors[investor];
        require(r.status == InvestorStatus.Pending, "InvestorIdentity: not pending");
        r.status          = InvestorStatus.Rejected;
        r.updatedAt       = block.timestamp;
        r.rejectionReason = reason;
        totalRejected++;
        emit InvestorRejected(investor, msg.sender, reason, block.timestamp);
    }

    /**
     * @notice Suspend a previously approved investor.
     */
    function suspendInvestor(address investor, string calldata reason)
        external
        onlyRole(KYC_ADMIN_ROLE)
    {
        require(investor != address(0), "InvestorIdentity: zero address");
        InvestorRecord storage r = _investors[investor];
        require(r.status == InvestorStatus.Approved, "InvestorIdentity: not approved");
        r.status          = InvestorStatus.Suspended;
        r.updatedAt       = block.timestamp;
        r.rejectionReason = reason;
        totalSuspended++;
        if (totalApproved > 0) totalApproved--;
        emit InvestorSuspended(investor, msg.sender, reason, block.timestamp);
    }

    /**
     * @notice Reinstate a suspended investor.
     */
    function reinstateInvestor(address investor) external onlyRole(KYC_ADMIN_ROLE) {
        require(investor != address(0), "InvestorIdentity: zero address");
        InvestorRecord storage r = _investors[investor];
        require(r.status == InvestorStatus.Suspended, "InvestorIdentity: not suspended");
        r.status    = InvestorStatus.Approved;
        r.updatedAt = block.timestamp;
        totalApproved++;
        if (totalSuspended > 0) totalSuspended--;
        emit InvestorReinstated(investor, msg.sender, block.timestamp);
    }

    // ─── View Functions ──────────────────────────────────────────────────────

    function isApproved(address investor) external view returns (bool) {
        return _investors[investor].status == InvestorStatus.Approved;
    }

    function getInvestorRecord(address investor) external view returns (InvestorRecord memory) {
        return _investors[investor];
    }

    function getInvestorStatus(address investor) external view returns (InvestorStatus) {
        return _investors[investor].status;
    }

    function getInvestors(uint256 offset, uint256 limit) external view returns (address[] memory) {
        uint256 total = _investorList.length;
        if (offset >= total) return new address[](0);
        uint256 end = offset + limit > total ? total : offset + limit;
        address[] memory result = new address[](end - offset);
        for (uint256 i = offset; i < end; i++) {
            result[i - offset] = _investorList[i];
        }
        return result;
    }

    function totalInvestors() external view returns (uint256) {
        return _investorList.length;
    }
}

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@openzeppelin/contracts/access/AccessControl.sol";

/**
 * @title BondAssetRegistry
 * @author Propx Protocol
 * @notice Registers and manages real-world bond information on-chain.
 *         Each bond must be approved by an admin before tokens can be issued.
 *
 * @dev PROTOTYPE NOTICE:
 *      Bond data is self-reported by issuers. The protocol does not independently
 *      verify the existence of the underlying real-world bond instrument.
 *      Reserve verification is handled separately by ReserveVerifier.sol.
 */
contract BondAssetRegistry is AccessControl {
    bytes32 public constant REGISTRY_ADMIN_ROLE = keccak256("REGISTRY_ADMIN_ROLE");
    bytes32 public constant ISSUER_ROLE         = keccak256("ISSUER_ROLE");

    enum BondStatus {
        None,       // 0: Does not exist
        Pending,    // 1: Registered, awaiting admin approval
        Approved,   // 2: Approved – tokens may be issued
        Rejected,   // 3: Rejected by admin
        Matured,    // 4: Bond has reached maturity
        Cancelled   // 5: Cancelled
    }

    struct Bond {
        string   bondId;
        string   name;
        string   description;
        address  issuer;
        uint256  faceValue;       // Total face value in mINR (18 decimals)
        uint256  couponRateBps;   // Annual coupon rate in basis points (800 = 8%)
        uint256  maturityDate;    // Unix timestamp of maturity
        uint256  tokenPrice;      // Price per token in mINR (18 decimals)
        uint256  totalSupply;     // Total tokens to issue
        uint256  availableSupply; // Tokens not yet minted
        uint256  issuedSupply;    // Tokens already minted
        BondStatus status;
        uint256  registeredAt;
        uint256  approvedAt;
        uint256  updatedAt;
        string   rejectionReason;
    }

    mapping(string => Bond)        private _bonds;
    string[]                       private _bondIds;
    mapping(address => string[])   private _issuerBonds;

    uint256 public totalBondsRegistered;
    uint256 public totalBondsApproved;

    // ─── Events ──────────────────────────────────────────────────────────────
    event BondRegistered(string indexed bondId, string name, address indexed issuer, uint256 faceValue, uint256 totalSupply, uint256 timestamp);
    event BondApproved(string indexed bondId, address indexed approvedBy, uint256 timestamp);
    event BondRejected(string indexed bondId, address indexed rejectedBy, string reason, uint256 timestamp);
    event BondMatured(string indexed bondId, uint256 timestamp);
    event BondCancelled(string indexed bondId, address indexed cancelledBy, uint256 timestamp);
    event BondSupplyUpdated(string indexed bondId, uint256 issuedSupply, uint256 availableSupply);

    // ─── Constructor ─────────────────────────────────────────────────────────
    constructor(address admin) {
        require(admin != address(0), "BondAssetRegistry: admin is zero address");
        _grantRole(DEFAULT_ADMIN_ROLE,  admin);
        _grantRole(REGISTRY_ADMIN_ROLE, admin);
        _grantRole(ISSUER_ROLE,         admin);
    }

    // ─── Bond Registration ───────────────────────────────────────────────────

    /**
     * @notice Register a new bond for review.
     * @param bondId        Unique bond identifier (e.g. "BOND001")
     * @param name          Bond name
     * @param description   Bond description
     * @param faceValue     Total face value in mINR wei (18 decimals)
     * @param couponRateBps Annual coupon rate in basis points (800 = 8%)
     * @param maturityDate  Unix timestamp of maturity (must be future)
     * @param tokenPrice    Price per token in mINR wei
     * @param totalSupply_  Total number of tokens to issue
     */
    function registerBond(
        string calldata bondId,
        string calldata name,
        string calldata description,
        uint256 faceValue,
        uint256 couponRateBps,
        uint256 maturityDate,
        uint256 tokenPrice,
        uint256 totalSupply_
    ) external onlyRole(ISSUER_ROLE) {
        require(bytes(bondId).length > 0,                           "BondAssetRegistry: empty bondId");
        require(bytes(name).length > 0,                             "BondAssetRegistry: empty name");
        require(_bonds[bondId].status == BondStatus.None,           "BondAssetRegistry: bondId already exists");
        require(faceValue > 0,                                      "BondAssetRegistry: faceValue must be > 0");
        require(couponRateBps <= 10_000,                            "BondAssetRegistry: coupon rate > 100%");
        require(maturityDate > block.timestamp,                     "BondAssetRegistry: maturity must be future");
        require(tokenPrice > 0,                                     "BondAssetRegistry: tokenPrice must be > 0");
        require(totalSupply_ > 0,                                   "BondAssetRegistry: totalSupply must be > 0");
        require(faceValue == tokenPrice * totalSupply_,             "BondAssetRegistry: faceValue != tokenPrice * totalSupply");

        _bonds[bondId] = Bond({
            bondId:          bondId,
            name:            name,
            description:     description,
            issuer:          msg.sender,
            faceValue:       faceValue,
            couponRateBps:   couponRateBps,
            maturityDate:    maturityDate,
            tokenPrice:      tokenPrice,
            totalSupply:     totalSupply_,
            availableSupply: totalSupply_,
            issuedSupply:    0,
            status:          BondStatus.Pending,
            registeredAt:    block.timestamp,
            approvedAt:      0,
            updatedAt:       block.timestamp,
            rejectionReason: ""
        });
        _bondIds.push(bondId);
        _issuerBonds[msg.sender].push(bondId);
        totalBondsRegistered++;

        emit BondRegistered(bondId, name, msg.sender, faceValue, totalSupply_, block.timestamp);
    }

    // ─── Admin Operations ────────────────────────────────────────────────────

    function approveBond(string calldata bondId) external onlyRole(REGISTRY_ADMIN_ROLE) {
        Bond storage b = _bonds[bondId];
        require(b.status == BondStatus.Pending, "BondAssetRegistry: bond not pending");
        b.status     = BondStatus.Approved;
        b.approvedAt = block.timestamp;
        b.updatedAt  = block.timestamp;
        totalBondsApproved++;
        emit BondApproved(bondId, msg.sender, block.timestamp);
    }

    function rejectBond(string calldata bondId, string calldata reason) external onlyRole(REGISTRY_ADMIN_ROLE) {
        Bond storage b = _bonds[bondId];
        require(b.status == BondStatus.Pending, "BondAssetRegistry: bond not pending");
        b.status          = BondStatus.Rejected;
        b.updatedAt       = block.timestamp;
        b.rejectionReason = reason;
        emit BondRejected(bondId, msg.sender, reason, block.timestamp);
    }

    function markMatured(string calldata bondId) external onlyRole(REGISTRY_ADMIN_ROLE) {
        Bond storage b = _bonds[bondId];
        require(b.status == BondStatus.Approved,          "BondAssetRegistry: bond not approved");
        require(block.timestamp >= b.maturityDate,         "BondAssetRegistry: bond not yet matured");
        b.status    = BondStatus.Matured;
        b.updatedAt = block.timestamp;
        emit BondMatured(bondId, block.timestamp);
    }

    function updateSupply(string calldata bondId, uint256 newIssuedSupply) external onlyRole(REGISTRY_ADMIN_ROLE) {
        Bond storage b = _bonds[bondId];
        require(b.status == BondStatus.Approved,                "BondAssetRegistry: bond not approved");
        require(newIssuedSupply <= b.totalSupply,               "BondAssetRegistry: exceeds total supply");
        b.issuedSupply    = newIssuedSupply;
        b.availableSupply = b.totalSupply - newIssuedSupply;
        emit BondSupplyUpdated(bondId, newIssuedSupply, b.availableSupply);
    }

    // ─── View Functions ──────────────────────────────────────────────────────

    function getBond(string calldata bondId) external view returns (Bond memory) {
        require(_bonds[bondId].status != BondStatus.None, "BondAssetRegistry: bond not found");
        return _bonds[bondId];
    }

    function getBondStatus(string calldata bondId) external view returns (BondStatus) {
        return _bonds[bondId].status;
    }

    function isBondApproved(string calldata bondId) external view returns (bool) {
        return _bonds[bondId].status == BondStatus.Approved;
    }

    function getBondIssuer(string calldata bondId) external view returns (address) {
        return _bonds[bondId].issuer;
    }

    function getBondMaturityDate(string calldata bondId) external view returns (uint256) {
        return _bonds[bondId].maturityDate;
    }

    function getBondTokenPrice(string calldata bondId) external view returns (uint256) {
        return _bonds[bondId].tokenPrice;
    }

    function getAllBondIds() external view returns (string[] memory) {
        return _bondIds;
    }

    function getBondsPaginated(uint256 offset, uint256 limit) external view returns (Bond[] memory) {
        uint256 total = _bondIds.length;
        if (offset >= total) return new Bond[](0);
        uint256 end = offset + limit > total ? total : offset + limit;
        Bond[] memory result = new Bond[](end - offset);
        for (uint256 i = offset; i < end; i++) {
            result[i - offset] = _bonds[_bondIds[i]];
        }
        return result;
    }

    function getIssuerBonds(address issuer) external view returns (string[] memory) {
        return _issuerBonds[issuer];
    }

    function totalBonds() external view returns (uint256) {
        return _bondIds.length;
    }
}

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@openzeppelin/contracts/access/AccessControl.sol";

/**
 * @title ReserveVerifier
 * @author Propx Protocol
 * @notice Verifies that a bond has sufficient reported reserves before tokens can be issued.
 *
 * @dev PROTOTYPE DISCLAIMER:
 *      This contract implements a MOCK reserve attestation mechanism.
 *      Reserve values are self-reported by an authorised oracle/admin role.
 *      This contract does NOT independently prove the existence of real-world assets.
 *
 *      In production, integrate a Chainlink Proof of Reserve oracle, a regulated
 *      custodian attestation service, or an equivalent trust-minimised mechanism.
 *
 *      Reserve Coverage Ratio:
 *         coverageRatioBps = (reportedReserve * 10_000) / issuanceValue
 *      A ratio >= 10_000 (100%) means reserves are considered sufficient.
 *
 *      Stale reports: Reports older than MAX_REPORT_AGE (7 days) are rejected.
 */
contract ReserveVerifier is AccessControl {
    bytes32 public constant RESERVE_ADMIN_ROLE = keccak256("RESERVE_ADMIN_ROLE");
    bytes32 public constant ORACLE_ROLE        = keccak256("ORACLE_ROLE");

    uint256 public constant MIN_COVERAGE_BPS = 10_000; // 100%
    uint256 public constant MAX_REPORT_AGE   = 7 days;

    struct ReserveReport {
        string  bondId;
        uint256 reportedReserve;    // Reported reserve in mINR (18 decimals)
        uint256 issuanceValue;      // Total token issuance value in mINR
        uint256 coverageRatioBps;   // reportedReserve * 10000 / issuanceValue
        bool    isSufficient;       // coverageRatioBps >= MIN_COVERAGE_BPS
        uint256 reportedAt;
        address reportedBy;
        bool    isVerified;         // Admin has explicitly verified
    }

    mapping(string => ReserveReport) private _reports;
    string[]                         private _reportedBondIds;

    // ─── Events ──────────────────────────────────────────────────────────────
    event ReserveReported(
        string indexed bondId,
        uint256 reportedReserve,
        uint256 issuanceValue,
        uint256 coverageRatioBps,
        bool isSufficient,
        address indexed reportedBy,
        uint256 timestamp
    );
    event ReserveVerified(string indexed bondId, address indexed verifiedBy, uint256 timestamp);

    // ─── Constructor ─────────────────────────────────────────────────────────
    constructor(address admin) {
        require(admin != address(0), "ReserveVerifier: admin is zero address");
        _grantRole(DEFAULT_ADMIN_ROLE,  admin);
        _grantRole(RESERVE_ADMIN_ROLE,  admin);
        _grantRole(ORACLE_ROLE,         admin);
    }

    // ─── Report Submission ───────────────────────────────────────────────────

    /**
     * @notice Submit or update a reserve report for a bond.
     * @param bondId          Bond identifier
     * @param reportedReserve Reported reserve value in mINR (18 decimals)
     * @param issuanceValue   Total value being tokenized in mINR (18 decimals)
     */
    function reportReserve(
        string calldata bondId,
        uint256 reportedReserve,
        uint256 issuanceValue
    ) external onlyRole(ORACLE_ROLE) {
        require(bytes(bondId).length > 0,  "ReserveVerifier: empty bondId");
        require(reportedReserve > 0,       "ReserveVerifier: reserve must be > 0");
        require(issuanceValue > 0,         "ReserveVerifier: issuance value must be > 0");

        uint256 coverage = (reportedReserve * 10_000) / issuanceValue;
        bool sufficient  = coverage >= MIN_COVERAGE_BPS;
        bool isNew       = bytes(_reports[bondId].bondId).length == 0;

        _reports[bondId] = ReserveReport({
            bondId:           bondId,
            reportedReserve:  reportedReserve,
            issuanceValue:    issuanceValue,
            coverageRatioBps: coverage,
            isSufficient:     sufficient,
            reportedAt:       block.timestamp,
            reportedBy:       msg.sender,
            isVerified:       false
        });

        if (isNew) _reportedBondIds.push(bondId);

        emit ReserveReported(bondId, reportedReserve, issuanceValue, coverage, sufficient, msg.sender, block.timestamp);
    }

    /**
     * @notice Admin explicitly verifies a reserve report (after off-chain review).
     */
    function verifyReserve(string calldata bondId) external onlyRole(RESERVE_ADMIN_ROLE) {
        ReserveReport storage r = _reports[bondId];
        require(bytes(r.bondId).length > 0,         "ReserveVerifier: no report for bond");
        require(r.isSufficient,                     "ReserveVerifier: coverage insufficient");
        require(!_isStale(r.reportedAt),            "ReserveVerifier: report is stale");
        r.isVerified = true;
        emit ReserveVerified(bondId, msg.sender, block.timestamp);
    }

    // ─── Eligibility Check ───────────────────────────────────────────────────

    /**
     * @notice Check whether minting is currently permitted for this bond.
     * @return eligible True if all checks pass
     * @return reason   Human-readable reason if not eligible
     */
    function isMintingEligible(string calldata bondId)
        external
        view
        returns (bool eligible, string memory reason)
    {
        ReserveReport storage r = _reports[bondId];
        if (bytes(r.bondId).length == 0)  return (false, "No reserve report submitted");
        if (!r.isSufficient)              return (false, "Reserve coverage below 100%");
        if (_isStale(r.reportedAt))       return (false, "Reserve report is stale (>7 days)");
        if (!r.isVerified)                return (false, "Reserve not yet verified by admin");
        return (true, "Reserve verified");
    }

    // ─── View Functions ──────────────────────────────────────────────────────

    function getReport(string calldata bondId) external view returns (ReserveReport memory) {
        return _reports[bondId];
    }

    function isReserveVerified(string calldata bondId) external view returns (bool) {
        ReserveReport storage r = _reports[bondId];
        return r.isVerified && r.isSufficient && !_isStale(r.reportedAt);
    }

    function getCoverageRatio(string calldata bondId) external view returns (uint256) {
        return _reports[bondId].coverageRatioBps;
    }

    function getAllReportedBondIds() external view returns (string[] memory) {
        return _reportedBondIds;
    }

    function _isStale(uint256 reportedAt) internal view returns (bool) {
        return block.timestamp > reportedAt + MAX_REPORT_AGE;
    }
}

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import "@openzeppelin/contracts/access/AccessControl.sol";
import "@openzeppelin/contracts/utils/ReentrancyGuard.sol";
import "./BondAssetRegistry.sol";
import "./InvestorIdentity.sol";

contract BondToken is ERC20, AccessControl, ReentrancyGuard {
    bytes32 public constant MINTER_ROLE = keccak256("MINTER_ROLE");
    bytes32 public constant BURNER_ROLE = keccak256("BURNER_ROLE");
    bytes32 public constant TOKEN_ADMIN_ROLE = keccak256("TOKEN_ADMIN_ROLE");

    string public bondId;
    uint256 public maxSupply;
    bool public transfersPaused;

    mapping(address => bool) public trustedContracts;

    BondAssetRegistry public immutable registry;
    InvestorIdentity public immutable identity;
    ReserveVerifier public immutable reserveVerifier;

    event TokensMinted(address indexed to, uint256 amount, address indexed mintedBy);
    event TokensBurned(address indexed from, uint256 amount, address indexed burnedBy);
    event TransfersPaused(address indexed by);
    event TransfersUnpaused(address indexed by);
    event TrustedContractUpdated(address indexed contractAddress, bool trusted);

    constructor(
        string memory _bondId,
        string memory _name,
        string memory _symbol,
        uint256 _maxSupply,
        address _registry,

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@openzeppelin/contracts/access/AccessControl.sol";
import "@openzeppelin/contracts/utils/ReentrancyGuard.sol";
import "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import "./BondToken.sol";
import "./InvestorIdentity.sol";
import "./BondAssetRegistry.sol";

/**
 * @title BondMarketplace
 * @author Propx Protocol
 * @notice Secondary marketplace for buying and selling bond tokens between approved investors.
 *
 * @dev Flow:
 *      1. Seller approves marketplace to spend their BondTokens.
 *      2. Seller calls createListing() — tokens are escrowed in this contract.
 *      3. Buyer approves marketplace to spend their mINR payment tokens.
 *      4. Buyer calls purchaseTokens() — payment flows to seller (minus platform fee),
 *         bond tokens flow to buyer.
 *      5. Seller may cancel an active listing to retrieve escrowed tokens.
 *
 * @dev PROTOTYPE NOTICE:
 *      Payment is in MockPaymentToken (mINR). Not real INR or any regulated currency.
 *      A 0.5% platform fee is charged on each trade (configurable by admin, max 5%).
 */
contract BondMarketplace is AccessControl, ReentrancyGuard {
    bytes32 public constant MARKETPLACE_ADMIN_ROLE = keccak256("MARKETPLACE_ADMIN_ROLE");

    enum ListingStatus { Active, Filled, Cancelled, Expired }

    struct Listing {
        uint256       listingId;
        address       seller;
        address       bondTokenAddress;
        string        bondId;
        uint256       quantity;
        uint256       remainingQuantity;
        uint256       pricePerToken;    // mINR wei per token
        uint256       createdAt;
        uint256       expiresAt;
        ListingStatus status;
    }

    struct Trade {
        uint256 tradeId;
        uint256 listingId;
        address buyer;
        address seller;
        string  bondId;
        uint256 quantity;
        uint256 pricePerToken;
        uint256 totalCost;
        uint256 timestamp;
    }

    IERC20            public immutable paymentToken;
    InvestorIdentity  public immutable identity;
    BondAssetRegistry public immutable registry;

    uint256 private _nextListingId;
    uint256 private _nextTradeId;

    uint256 public platformFeeBps; // default 50 = 0.5%
    address public feeRecipient;

    mapping(uint256 => Listing) private _listings;
    mapping(uint256 => Trade)   private _trades;
    uint256[]                   private _activeListingIds;
    uint256[]                   private _allListingIds;
    mapping(address => uint256[]) private _sellerListings;
    mapping(address => uint256[]) private _buyerTrades;

    // ─── Events ──────────────────────────────────────────────────────────────
    event ListingCreated(uint256 indexed listingId, address indexed seller, string bondId, uint256 quantity, uint256 pricePerToken, uint256 expiresAt);
    event TokensPurchased(uint256 indexed tradeId, uint256 indexed listingId, address indexed buyer, address seller, string bondId, uint256 quantity, uint256 totalCost);
    event ListingCancelled(uint256 indexed listingId, address indexed by);
    event PlatformFeeUpdated(uint256 newFeeBps);

    // ─── Constructor ─────────────────────────────────────────────────────────
    constructor(
        address _paymentToken,
        address _identity,
        address _registry,
        address _admin
    ) {
        require(_paymentToken != address(0), "BondMarketplace: payment token is zero");
        require(_identity     != address(0), "BondMarketplace: identity is zero");
        require(_registry     != address(0), "BondMarketplace: registry is zero");
        require(_admin        != address(0), "BondMarketplace: admin is zero");

        paymentToken   = IERC20(_paymentToken);
        identity       = InvestorIdentity(_identity);
        registry       = BondAssetRegistry(_registry);
        platformFeeBps = 50;   // 0.5%
        feeRecipient   = _admin;

        _grantRole(DEFAULT_ADMIN_ROLE,       _admin);
        _grantRole(MARKETPLACE_ADMIN_ROLE,   _admin);
    }

    // ─── Listing ─────────────────────────────────────────────────────────────

    /**
     * @notice Create a sell listing. Tokens are escrowed in this contract.
     * @param bondTokenAddress  Address of BondToken contract
     * @param bondId            Bond identifier
     * @param quantity          Number of tokens to sell
     * @param pricePerToken     Price per token in mINR wei
     * @param duration          Duration in seconds (max 90 days)
     */
    function createListing(
        address bondTokenAddress,
        string calldata bondId,
        uint256 quantity,
        uint256 pricePerToken,
        uint256 duration
    ) external nonReentrant returns (uint256) {
        require(identity.isApproved(msg.sender),    "BondMarketplace: seller not approved");
        require(bondTokenAddress != address(0),     "BondMarketplace: token address zero");
        require(quantity > 0,                       "BondMarketplace: quantity must be > 0");
        require(pricePerToken > 0,                  "BondMarketplace: price must be > 0");
        require(duration > 0 && duration <= 90 days,"BondMarketplace: invalid duration");

        // Bond must be active
        require(
            registry.getBondStatus(bondId) == BondAssetRegistry.BondStatus.Approved,
            "BondMarketplace: bond not active"
        );

        BondToken bt = BondToken(bondTokenAddress);
        require(bt.balanceOf(msg.sender) >= quantity, "BondMarketplace: insufficient token balance");

        // Escrow tokens
        require(
            bt.transferFrom(msg.sender, address(this), quantity),
            "BondMarketplace: token escrow failed"
        );

        uint256 listingId = _nextListingId++;
        uint256 expiresAt = block.timestamp + duration;

        _listings[listingId] = Listing({
            listingId:         listingId,
            seller:            msg.sender,
            bondTokenAddress:  bondTokenAddress,
            bondId:            bondId,
            quantity:          quantity,
            remainingQuantity: quantity,
            pricePerToken:     pricePerToken,
            createdAt:         block.timestamp,
            expiresAt:         expiresAt,
            status:            ListingStatus.Active
        });

        _allListingIds.push(listingId);
        _activeListingIds.push(listingId);
        _sellerListings[msg.sender].push(listingId);

        emit ListingCreated(listingId, msg.sender, bondId, quantity, pricePerToken, expiresAt);
        return listingId;
    }

    // ─── Purchasing ──────────────────────────────────────────────────────────

    /**
     * @notice Purchase tokens from an active listing.
     * @param listingId Listing to buy from
     * @param quantity  Number of tokens to buy
     */
    function purchaseTokens(uint256 listingId, uint256 quantity) external nonReentrant {
        require(identity.isApproved(msg.sender), "BondMarketplace: buyer not approved");

        Listing storage l = _listings[listingId];
        require(l.status == ListingStatus.Active,            "BondMarketplace: listing not active");
        require(block.timestamp <= l.expiresAt,              "BondMarketplace: listing expired");
        require(quantity > 0 && quantity <= l.remainingQuantity, "BondMarketplace: invalid quantity");
        require(msg.sender != l.seller,                      "BondMarketplace: cannot buy own listing");

        uint256 totalCost      = l.pricePerToken * quantity;
        uint256 platformFee    = (totalCost * platformFeeBps) / 10_000;
        uint256 sellerProceeds = totalCost - platformFee;

        // Pull payment from buyer
        require(
            paymentToken.transferFrom(msg.sender, address(this), totalCost),
            "BondMarketplace: payment transfer failed"
        );

        // Pay seller
        require(paymentToken.transfer(l.seller, sellerProceeds), "BondMarketplace: seller payment failed");

        // Pay platform fee
        if (platformFee > 0) {
            require(paymentToken.transfer(feeRecipient, platformFee), "BondMarketplace: fee transfer failed");
        }

        // Deliver bond tokens to buyer
        BondToken bt = BondToken(l.bondTokenAddress);
        require(bt.transfer(msg.sender, quantity), "BondMarketplace: token delivery failed");

        // Update listing
        l.remainingQuantity -= quantity;
        if (l.remainingQuantity == 0) {
            l.status = ListingStatus.Filled;
            _removeActiveListing(listingId);
        }

        // Record trade
        uint256 tradeId = _nextTradeId++;
        _trades[tradeId] = Trade({
            tradeId:      tradeId,
            listingId:    listingId,
            buyer:        msg.sender,
            seller:       l.seller,
            bondId:       l.bondId,
            quantity:     quantity,
            pricePerToken:l.pricePerToken,
            totalCost:    totalCost,
            timestamp:    block.timestamp
        });
        _buyerTrades[msg.sender].push(tradeId);

        emit TokensPurchased(tradeId, listingId, msg.sender, l.seller, l.bondId, quantity, totalCost);
    }

    // ─── Cancellation ────────────────────────────────────────────────────────

    function cancelListing(uint256 listingId) external nonReentrant {
        Listing storage l = _listings[listingId];
        require(l.status == ListingStatus.Active, "BondMarketplace: not active");
        require(
            msg.sender == l.seller || hasRole(MARKETPLACE_ADMIN_ROLE, msg.sender),
            "BondMarketplace: not authorised"
        );
        l.status = ListingStatus.Cancelled;
        _removeActiveListing(listingId);

        // Return escrowed tokens to seller
        BondToken bt = BondToken(l.bondTokenAddress);
        require(bt.transfer(l.seller, l.remainingQuantity), "BondMarketplace: return failed");

        emit ListingCancelled(listingId, msg.sender);
    }

    // ─── Admin ───────────────────────────────────────────────────────────────

    function setPlatformFee(uint256 feeBps) external onlyRole(MARKETPLACE_ADMIN_ROLE) {
        require(feeBps <= 500, "BondMarketplace: fee too high (max 5%)");
        platformFeeBps = feeBps;
        emit PlatformFeeUpdated(feeBps);
    }

    function setFeeRecipient(address recipient) external onlyRole(MARKETPLACE_ADMIN_ROLE) {
        require(recipient != address(0), "BondMarketplace: zero address");
        feeRecipient = recipient;
    }

    // ─── View Functions ──────────────────────────────────────────────────────

    function getListing(uint256 listingId) external view returns (Listing memory) { return _listings[listingId]; }
    function getTrade(uint256 tradeId) external view returns (Trade memory)       { return _trades[tradeId]; }
    function getActiveListings() external view returns (uint256[] memory)         { return _activeListingIds; }
    function getAllListingIds() external view returns (uint256[] memory)           { return _allListingIds; }
    function getSellerListings(address seller) external view returns (uint256[] memory) { return _sellerListings[seller]; }
    function getBuyerTrades(address buyer) external view returns (uint256[] memory)     { return _buyerTrades[buyer]; }
    function totalListings() external view returns (uint256) { return _nextListingId; }
    function totalTrades() external view returns (uint256)   { return _nextTradeId; }

    // ─── Internal ────────────────────────────────────────────────────────────

    function _removeActiveListing(uint256 listingId) internal {
        uint256 len = _activeListingIds.length;
        for (uint256 i = 0; i < len; i++) {
            if (_activeListingIds[i] == listingId) {
                _activeListingIds[i] = _activeListingIds[len - 1];
                _activeListingIds.pop();
                break;
            }
        }
    }
}

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

import "@openzeppelin/contracts/access/AccessControl.sol";
import "@openzeppelin/contracts/utils/ReentrancyGuard.sol";
import "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import "./BondToken.sol";
import "./InvestorIdentity.sol";
import "./BondAssetRegistry.sol";

/**
 * @title CouponManager
 * @author Propx Protocol
 * @notice Handles coupon (interest) distributions and bond maturity redemption.
 *
 * @dev Coupon Distribution Flow:
 *      1. Admin/Issuer calls fundCouponPeriod() — provides mINR equal to
 *         (couponPerToken × totalTokenSupply) and specifies an optional claim deadline.
 *      2. Each approved token holder calls claimCoupon(periodId) to receive their share.
 *      3. Claim = balanceOf(investor) × couponPerToken.
 *      4. Duplicate claims for the same period are blocked.
 *
 * @dev Redemption Flow (at or after maturity):
 *      1. Issuer pre-funds the redemption pool via fundRedemptionPool().
 *      2. Each investor calls redeemBond(amount) — tokens are burned and
 *         face value (tokenPrice × amount) is paid in mINR.
 *      3. Early redemption is blocked.
 *
 * @dev Annual coupon example (illustrative):
 *      Face value = ₹1,00,00,000 | Total supply = 10,000 | Coupon rate = 8%
 *      Annual coupon pool = ₹80,00,000 (₹8,000 per 100-token holding)
 *      couponPerToken = ₹800 (in mINR wei = 800 * 10^18)
 */
contract CouponManager is AccessControl, ReentrancyGuard {
    bytes32 public constant COUPON_ADMIN_ROLE  = keccak256("COUPON_ADMIN_ROLE");
    bytes32 public constant ISSUER_ADMIN_ROLE  = keccak256("ISSUER_ADMIN_ROLE");

    struct CouponPeriod {
        uint256 periodId;
        string  bondId;
        uint256 couponPerToken;   // mINR per token (18 decimals)
        uint256 totalFunded;      // Total mINR deposited
        uint256 totalClaimed;     // Total mINR claimed so far
        uint256 snapshotTimestamp;
        uint256 claimDeadline;    // 0 = no deadline
        bool    isActive;
    }

    struct RedemptionRecord {
        address investor;
        string  bondId;
        uint256 tokensBurned;
        uint256 redemptionAmount;
        uint256 timestamp;
    }

    IERC20            public immutable paymentToken;
    BondToken         public immutable bondToken;
    InvestorIdentity  public immutable identity;
    BondAssetRegistry public immutable registry;
    string public bondId;

    uint256 private _nextPeriodId;

    mapping(uint256 => CouponPeriod)              private _periods;
    uint256[]                                     private _periodIds;
    mapping(address => mapping(uint256 => bool))  private _claimed;
    mapping(address => uint256)                   public totalCouponsReceived;

    RedemptionRecord[] private _redemptions;
    mapping(address => bool) public hasRedeemed;

    // ─── Events ──────────────────────────────────────────────────────────────
    event CouponPeriodFunded(uint256 indexed periodId, string bondId, uint256 couponPerToken, uint256 totalFunded, uint256 snapshotTimestamp);
    event CouponClaimed(uint256 indexed periodId, address indexed investor, uint256 amount, uint256 timestamp);
    event BondRedeemed(address indexed investor, string bondId, uint256 tokensBurned, uint256 redemptionAmount, uint256 timestamp);
    event RedemptionPoolFunded(address indexed by, uint256 amount, uint256 timestamp);

    // ─── Constructor ─────────────────────────────────────────────────────────
    constructor(
        string memory _bondId,
        address _paymentToken,
        address _bondToken,
        address _identity,
        address _registry,
        address _admin
    ) {
        require(bytes(_bondId).length > 0,  "CouponManager: empty bondId");
        require(_paymentToken != address(0),"CouponManager: payment token zero");
        require(_bondToken    != address(0),"CouponManager: bond token zero");
        require(_identity     != address(0),"CouponManager: identity zero");
        require(_registry     != address(0),"CouponManager: registry zero");
        require(_admin        != address(0),"CouponManager: admin zero");

        bondId       = _bondId;
        paymentToken = IERC20(_paymentToken);
        bondToken    = BondToken(_bondToken);
        identity     = InvestorIdentity(_identity);
        registry     = BondAssetRegistry(_registry);

        _grantRole(DEFAULT_ADMIN_ROLE,  _admin);
        _grantRole(COUPON_ADMIN_ROLE,   _admin);
        _grantRole(ISSUER_ADMIN_ROLE,   _admin);
    }

    // ─── Coupon Distribution ─────────────────────────────────────────────────

    /**
     * @notice Issuer/admin funds a coupon distribution period.
     *         Must first approve this contract to spend (couponPerToken × totalSupply) mINR.
     * @param couponPerToken Amount per token in mINR wei
     * @param claimDeadline  Unix deadline for claims (0 = no deadline)
     */
    function fundCouponPeriod(uint256 couponPerToken, uint256 claimDeadline)
        external
        onlyRole(ISSUER_ADMIN_ROLE)
        nonReentrant
    {
        require(couponPerToken > 0, "CouponManager: coupon must be > 0");
        uint256 supply = bondToken.totalSupply();
        require(supply > 0, "CouponManager: no tokens issued yet");
        uint256 totalFunded = couponPerToken * supply;

        require(
            paymentToken.transferFrom(msg.sender, address(this), totalFunded),
            "CouponManager: payment pull failed"
        );

        uint256 periodId = _nextPeriodId++;
        _periods[periodId] = CouponPeriod({
            periodId:          periodId,
            bondId:            bondId,
            couponPerToken:    couponPerToken,
            totalFunded:       totalFunded,
            totalClaimed:      0,
            snapshotTimestamp: block.timestamp,
            claimDeadline:     claimDeadline,
            isActive:          true
        });
        _periodIds.push(periodId);

        emit CouponPeriodFunded(periodId, bondId, couponPerToken, totalFunded, block.timestamp);
    }

    /**
     * @notice Investor claims their coupon for a period.
     * @param periodId The coupon period to claim
     */
    function claimCoupon(uint256 periodId) external nonReentrant {
        require(identity.isApproved(msg.sender),  "CouponManager: not approved investor");
        require(!_claimed[msg.sender][periodId],  "CouponManager: already claimed");

        CouponPeriod storage p = _periods[periodId];
        require(p.isActive,                       "CouponManager: period not active");
        require(
            p.claimDeadline == 0 || block.timestamp <= p.claimDeadline,
            "CouponManager: claim deadline passed"
        );

        uint256 balance = bondToken.balanceOf(msg.sender);
        require(balance > 0, "CouponManager: no tokens held");

        uint256 claimAmount = balance * p.couponPerToken;
        require(
            p.totalClaimed + claimAmount <= p.totalFunded,
            "CouponManager: coupon pool exhausted"
        );

        _claimed[msg.sender][periodId] = true;
        p.totalClaimed += claimAmount;
        totalCouponsReceived[msg.sender] += claimAmount;

        require(
            paymentToken.transfer(msg.sender, claimAmount),
            "CouponManager: coupon transfer failed"
        );

        emit CouponClaimed(periodId, msg.sender, claimAmount, block.timestamp);
    }

    // ─── Redemption ──────────────────────────────────────────────────────────

    /**
     * @notice Redeem bond tokens at or after maturity.
     *         Tokens are burned and face value (tokenPrice × amount) paid in mINR.
     * @param amount Number of tokens to redeem
     */
    function redeemBond(uint256 amount) external nonReentrant {
        require(identity.isApproved(msg.sender), "CouponManager: not approved investor");
        require(amount > 0,                       "CouponManager: amount must be > 0");
        require(!hasRedeemed[msg.sender],         "CouponManager: already redeemed");

        uint256 maturityDate = registry.getBondMaturityDate(bondId);
        require(block.timestamp >= maturityDate,  "CouponManager: bond not yet matured");
        require(bondToken.balanceOf(msg.sender) >= amount, "CouponManager: insufficient tokens");

        uint256 tokenPrice = registry.getBondTokenPrice(bondId);
        uint256 redemptionAmount = tokenPrice * amount;

        hasRedeemed[msg.sender] = true;

        // Burn tokens (CouponManager must have BURNER_ROLE on BondToken)
        bondToken.burn(msg.sender, amount);

        require(
            paymentToken.transfer(msg.sender, redemptionAmount),
            "CouponManager: redemption payment failed"
        );

        _redemptions.push(RedemptionRecord({
            investor:         msg.sender,
            bondId:           bondId,
            tokensBurned:     amount,
            redemptionAmount: redemptionAmount,
            timestamp:        block.timestamp
        }));

        emit BondRedeemed(msg.sender, bondId, amount, redemptionAmount, block.timestamp);
    }

    // ─── Admin ───────────────────────────────────────────────────────────────

    /**
     * @notice Issuer pre-funds the redemption pool before maturity.
     */
    function fundRedemptionPool(uint256 amount) external onlyRole(ISSUER_ADMIN_ROLE) nonReentrant {
        require(amount > 0, "CouponManager: amount must be > 0");
        require(
            paymentToken.transferFrom(msg.sender, address(this), amount),
            "CouponManager: transfer failed"
        );
        emit RedemptionPoolFunded(msg.sender, amount, block.timestamp);
    }

    function deactivatePeriod(uint256 periodId) external onlyRole(COUPON_ADMIN_ROLE) {
        _periods[periodId].isActive = false;
    }

    // ─── View Functions ──────────────────────────────────────────────────────

    function getCouponPeriod(uint256 periodId) external view returns (CouponPeriod memory) {
        return _periods[periodId];
    }

    function getAllPeriodIds() external view returns (uint256[] memory) { return _periodIds; }

    function hasClaimed(address investor, uint256 periodId) external view returns (bool) {
        return _claimed[investor][periodId];
    }

    function getClaimableAmount(address investor, uint256 periodId) external view returns (uint256) {
        if (_claimed[investor][periodId]) return 0;
        CouponPeriod storage p = _periods[periodId];
        if (!p.isActive) return 0;
        return bondToken.balanceOf(investor) * p.couponPerToken;
    }

    function totalPeriods() external view returns (uint256) { return _nextPeriodId; }

    function getRedemptions() external view returns (RedemptionRecord[] memory) { return _redemptions; }

    function getContractBalance() external view returns (uint256) {
        return paymentToken.balanceOf(address(this));
    }
}
